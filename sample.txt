// Sample Python Code
def greet(name):
    """Function to greet a user"""
    return f"Hello, {name}!"

class Calculator:
    """Simple calculator class"""
    def add(self, a, b):
        return a + b
    
    def subtract(self, a, b):
        return a - b
    
    def multiply(self, a, b):
        return a * b

# Usage example
if __name__ == "__main__":
    print(greet("World"))
    calc = Calculator()
    print(f"5 + 3 = {calc.add(5, 3)}")
    print(f"10 - 4 = {calc.subtract(10, 4)}")
    print(f"6 * 7 = {calc.multiply(6, 7)}")
